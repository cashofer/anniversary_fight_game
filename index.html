    function ensureAudioReady() {

      if (!soundEnabled) return;

      try {

        if (!audioCtx) {

          const Ctx = window.AudioContext || window.webkitAudioContext;

          if (Ctx) {

            audioCtx = new Ctx();

          }

        }

        if (audioCtx && audioCtx.state === "suspended") {

          audioCtx.resume().catch(function () {});

        }

      } catch (err) {

        // 忽略受限裝置音效錯誤

      }

    }

    function playSynthTone(freq, duration, waveType, gainVal) {

      if (!soundEnabled) return;

      try {

        ensureAudioReady();

        if (!audioCtx) return;

        const t = audioCtx.currentTime;

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = waveType || "sine";

        osc.frequency.setValueAtTime(freq, t);

        gain.gain.setValueAtTime(gainVal || 0.1, t);

        gain.gain.exponentialRampToValueAtTime(0.001, t + duration);

        osc.connect(gain);

        gain.connect(audioCtx.destination);

        osc.start(t);

        osc.stop(t + duration + 0.01);

      } catch (err) {

        // 單次音效例外安全略過

      }

    }

    function bindPadControl(id, onPress, onRelease) {

      const el = document.getElementById(id);

      if (!el) return;

 

      el.addEventListener("pointerdown", function (ev) {

        ev.preventDefault();

        ensureAudioReady();

        el.classList.add("pressed");

        if (onPress) onPress();

      });

 

      const releaseFn = function (ev) {

        if (ev) ev.preventDefault();

        el.classList.remove("pressed");

        if (onRelease) onRelease();

      };

 

      el.addEventListener("pointerup", releaseFn);

      el.addEventListener("pointercancel", releaseFn);

      el.addEventListener("pointerleave", releaseFn);

    }

    function stepQueenMovement() {

      if (battleStatus !== "active") {

        queenPlayer.vx = 0;

        return;

      }

      const speed = 265;

      if (moveInput.left && !moveInput.right) {

        queenPlayer.vx = -speed;

        queenPlayer.facing = -1;

      } else if (moveInput.right && !moveInput.left) {

        queenPlayer.vx = speed;

        queenPlayer.facing = 1;

      } else {

        queenPlayer.vx = 0;

      }

    }

 

    function stepServantBrain(dt) {

      if (battleStatus !== "active") {

        servantAi.vx = 0;

        return;

      }

      const dx = queenPlayer.x - servantAi.x;

      servantAi.facing = dx >= 0 ? 1 : -1;

 

      const moveSpd = aiDifficulty === "survival" ? 182 : 108;

      const atkChance = aiDifficulty === "survival" ? 1.9 : 0.9;

 

      if (Math.abs(dx) > 105) {

        servantAi.vx = servantAi.facing * moveSpd;

      } else {

        servantAi.vx = 0;

        if (Math.random() < dt * atkChance) {

          servantAi.performAttack(Math.random() < 0.65 ? "servantStrike" : "card", queenPlayer);

        }

      }

 

      if (Math.random() < dt * 0.55) {

        servantAi.jumpUnit();

      }

    }

 

    function paintWorld(dt) {

      ctx.save();

      const sx = canvas.width / WORLD_W;

      const sy = canvas.height / WORLD_H;

      ctx.scale(sx, sy);

 

      if (screenShakeSec > 0) {

        const mag = screenShakeSec * 18;

        ctx.translate((Math.random() - 0.5) * mag, (Math.random() - 0.5) * mag);

      }

 

      drawSceneLayer(ctx);

      kidReina.drawKid(ctx);

      kidEian.drawKid(ctx);

      queenPlayer.drawUnit(ctx);

      servantAi.drawUnit(ctx);

      updateAndDrawEffects(ctx, dt);

 

      if (screenFlashSec > 0) {

        ctx.fillStyle = `rgba(239, 68, 68, ${Math.min(0.45, screenFlashSec)})`;

        ctx.fillRect(0, 0, WORLD_W, WORLD_H);

      }

 

      ctx.font = "700 11px monospace";

      ctx.textAlign = "right";

      ctx.fillStyle = "rgba(15, 23, 42, 0.7)";

      ctx.fillText(`FRAME #${frameCounter}`, WORLD_W - 10, 20);

 

      ctx.restore();

    }

 

    function loopFrame(ts) {

      try {

        if (!lastTimestamp) lastTimestamp = ts;

        const dt = Math.max(0, Math.min(0.05, (ts - lastTimestamp) / 1000));

        lastTimestamp = ts;

        frameCounter++;

 

        if (screenShakeSec > 0) screenShakeSec = Math.max(0, screenShakeSec - dt);

        if (screenFlashSec > 0) screenFlashSec = Math.max(0, screenFlashSec - dt);

        if (mutationBannerSec > 0) mutationBannerSec = Math.max(0, mutationBannerSec - dt);

 

        if (battleStatus === "active") {

          stepQueenMovement();

          stepServantBrain(dt);

 

          timeAcc += dt;

          if (timeAcc >= 1) {

            timeAcc -= 1;

            remainingTime = Math.max(0, remainingTime - 1);

            if (remainingTime <= 0) {

              openKoModal();

            }

          }

        }

 

        queenPlayer.updateUnit(dt, servantAi);

        servantAi.updateUnit(dt, queenPlayer);

        kidEian.updateKid(dt, queenPlayer, servantAi);

        kidReina.updateKid(dt, queenPlayer, servantAi);

 

        paintWorld(dt);

        syncVitalsHud();

      } catch (err) {

        // 單幀例外安全略過

      }

      window.requestAnimationFrame(loopFrame);

    }
